## 方法 1：基础星号图案

### 直角三角形星号图案

```java
import java.util.Scanner;

public class BasicStarPatterns {

    // 直角三角形图案
    public static void printRightTriangle(int n) {
        System.out.println("Right Triangle Pattern:");
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= i; j++) {
                System.out.print("* ");
            }
            System.out.println();
        }
    }

    // 倒置直角三角形
    public static void printInvertedRightTriangle(int n) {
        System.out.println("\nInverted Right Triangle Pattern:");
        for (int i = n; i >= 1; i--) {
            for (int j = 1; j <= i; j++) {
                System.out.print("* ");
            }
            System.out.println();
        }
    }

    // 左三角形（右对齐）
    public static void printLeftTriangle(int n) {
        System.out.println("\nLeft Triangle Pattern:");
        for (int i = 1; i <= n; i++) {
            // 打印空格
            for (int j = 1; j <= n - i; j++) {
                System.out.print("  ");
            }
            // 打印星号
            for (int j = 1; j <= i; j++) {
                System.out.print("* ");
            }
            System.out.println();
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        System.out.print("Enter the number of rows: ");
        int rows = scanner.nextInt();

        printRightTriangle(rows);
        printInvertedRightTriangle(rows);
        printLeftTriangle(rows);

        scanner.close();
    }
}
```

## 方法 2：金字塔与菱形图案

```java
import java.util.Scanner;

public class PyramidPatterns {

    // 完整金字塔（三角形）
    public static void printPyramid(int n) {
        System.out.println("Pyramid Pattern:");
        for (int i = 1; i <= n; i++) {
            // 打印空格
            for (int j = 1; j <= n - i; j++) {
                System.out.print(" ");
            }
            // 打印星号
            for (int j = 1; j <= 2 * i - 1; j++) {
                System.out.print("*");
            }
            System.out.println();
        }
    }

    // 倒置金字塔
    public static void printInvertedPyramid(int n) {
        System.out.println("\nInverted Pyramid Pattern:");
        for (int i = n; i >= 1; i--) {
            // 打印空格
            for (int j = 1; j <= n - i; j++) {
                System.out.print(" ");
            }
            // 打印星号
            for (int j = 1; j <= 2 * i - 1; j++) {
                System.out.print("*");
            }
            System.out.println();
        }
    }

    // 菱形图案
    public static void printDiamond(int n) {
        System.out.println("\nDiamond Pattern:");
        // 上半部分（包含中间行）
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n - i; j++) {
                System.out.print(" ");
            }
            for (int j = 1; j <= 2 * i - 1; j++) {
                System.out.print("*");
            }
            System.out.println();
        }
        // 下半部分
        for (int i = n - 1; i >= 1; i--) {
            for (int j = 1; j <= n - i; j++) {
                System.out.print(" ");
            }
            for (int j = 1; j <= 2 * i - 1; j++) {
                System.out.print("*");
            }
            System.out.println();
        }
    }

    // 空心菱形
    public static void printHollowDiamond(int n) {
        System.out.println("\nHollow Diamond Pattern:");
        // 上半部分
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n - i; j++) {
                System.out.print(" ");
            }
            for (int j = 1; j <= 2 * i - 1; j++) {
                if (j == 1 || j == 2 * i - 1) {
                    System.out.print("*");
                } else {
                    System.out.print(" ");
                }
            }
            System.out.println();
        }
        // 下半部分
        for (int i = n - 1; i >= 1; i--) {
            for (int j = 1; j <= n - i; j++) {
                System.out.print(" ");
            }
            for (int j = 1; j <= 2 * i - 1; j++) {
                if (j == 1 || j == 2 * i - 1) {
                    System.out.print("*");
                } else {
                    System.out.print(" ");
                }
            }
            System.out.println();
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        System.out.print("Enter the number of rows: ");
        int rows = scanner.nextInt();

        printPyramid(rows);
        printInvertedPyramid(rows);
        printDiamond(rows);
        printHollowDiamond(rows);

        scanner.close();
    }
}
```

## 方法 3：数字图案

```java
import java.util.Scanner;

public class NumberPatterns {

    // 简单数字三角形
    public static void printNumberTriangle(int n) {
        System.out.println("Number Triangle:");
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= i; j++) {
                System.out.print(j + " ");
            }
            System.out.println();
        }
    }

    // 帕斯卡三角形
    public static void printPascalTriangle(int n) {
        System.out.println("\nPascal's Triangle:");
        for (int i = 0; i < n; i++) {
            // 打印用于对齐的空格
            for (int j = 0; j < n - i - 1; j++) {
                System.out.print(" ");
            }

            int number = 1;
            for (int j = 0; j <= i; j++) {
                System.out.print(number + " ");
                number = number * (i - j) / (j + 1);
            }
            System.out.println();
        }
    }

    // 弗洛伊德三角形
    public static void printFloydTriangle(int n) {
        System.out.println("\nFloyd's Triangle:");
        int number = 1;
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= i; j++) {
                System.out.print(number + " ");
                number++;
            }
            System.out.println();
        }
    }

    // 数字金字塔
    public static void printNumberPyramid(int n) {
        System.out.println("\nNumber Pyramid:");
        for (int i = 1; i <= n; i++) {
            // 打印空格
            for (int j = 1; j <= n - i; j++) {
                System.out.print(" ");
            }
            // 打印递增数字
            for (int j = 1; j <= i; j++) {
                System.out.print(j);
            }
            // 打印递减数字
            for (int j = i - 1; j >= 1; j--) {
                System.out.print(j);
            }
            System.out.println();
        }
    }

    // 二进制图案
    public static void printBinaryPattern(int n) {
        System.out.println("\nBinary Pattern:");
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= i; j++) {
                System.out.print((i + j) % 2 + " ");
            }
            System.out.println();
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        System.out.print("Enter the number of rows: ");
        int rows = scanner.nextInt();

        printNumberTriangle(rows);
        printPascalTriangle(rows);
        printFloydTriangle(rows);
        printNumberPyramid(rows);
        printBinaryPattern(rows);

        scanner.close();
    }
}
```

## 方法 4：字母图案

```java
import java.util.Scanner;

public class AlphabetPatterns {

    // 简单字母三角形
    public static void printAlphabetTriangle(int n) {
        System.out.println("Alphabet Triangle:");
        for (int i = 1; i <= n; i++) {
            char ch = 'A';
            for (int j = 1; j <= i; j++) {
                System.out.print(ch + " ");
                ch++;
            }
            System.out.println();
        }
    }

    // 字母金字塔
    public static void printAlphabetPyramid(int n) {
        System.out.println("\nAlphabet Pyramid:");
        for (int i = 1; i <= n; i++) {
            // 打印空格
            for (int j = 1; j <= n - i; j++) {
                System.out.print(" ");
            }

            // 打印递增字母
            char ch = 'A';
            for (int j = 1; j <= i; j++) {
                System.out.print(ch);
                ch++;
            }

            // 打印递减字母
            ch = (char)('A' + i - 2);
            for (int j = 1; j <= i - 1; j++) {
                System.out.print(ch);
                ch--;
            }

            System.out.println();
        }
    }

    // 重复字母图案
    public static void printRepeatingAlphabet(int n) {
        System.out.println("\nRepeating Alphabet Pattern:");
        for (int i = 1; i <= n; i++) {
            char ch = (char)('A' + i - 1);
            for (int j = 1; j <= i; j++) {
                System.out.print(ch + " ");
            }
            System.out.println();
        }
    }

    // 字母菱形
    public static void printAlphabetDiamond(int n) {
        System.out.println("\nAlphabet Diamond:");
        // 上半部分
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= n - i; j++) {
                System.out.print(" ");
            }
            char ch = 'A';
            for (int j = 1; j <= i; j++) {
                System.out.print(ch);
                ch++;
            }
            ch = (char)(ch - 2);
            for (int j = 1; j <= i - 1; j++) {
                System.out.print(ch);
                ch--;
            }
            System.out.println();
        }

        // 下半部分
        for (int i = n - 1; i >= 1; i--) {
            for (int j = 1; j <= n - i; j++) {
                System.out.print(" ");
            }
            char ch = 'A';
            for (int j = 1; j <= i; j++) {
                System.out.print(ch);
                ch++;
            }
            ch = (char)(ch - 2);
            for (int j = 1; j <= i - 1; j++) {
                System.out.print(ch);
                ch--;
            }
            System.out.println();
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        System.out.print("Enter the number of rows: ");
        int rows = scanner.nextInt();

        printAlphabetTriangle(rows);
        printAlphabetPyramid(rows);
        printRepeatingAlphabet(rows);
        printAlphabetDiamond(rows);

        scanner.close();
    }
}
```

## 方法 5：进阶几何图案

```java
import java.util.Scanner;

public class AdvancedPatterns {

    // 蝴蝶图案
    public static void printButterfly(int n) {
        System.out.println("Butterfly Pattern:");
        // 上半部分
        for (int i = 1; i <= n; i++) {
            // 左侧星号
            for (int j = 1; j <= i; j++) {
                System.out.print("*");
            }
            // 中间空格
            for (int j = 1; j <= 2 * (n - i); j++) {
                System.out.print(" ");
            }
            // 右侧星号
            for (int j = 1; j <= i; j++) {
                System.out.print("*");
            }
            System.out.println();
        }

        // 下半部分
        for (int i = n - 1; i >= 1; i--) {
            // 左侧星号
            for (int j = 1; j <= i; j++) {
                System.out.print("*");
            }
            // 中间空格
            for (int j = 1; j <= 2 * (n - i); j++) {
                System.out.print(" ");
            }
            // 右侧星号
            for (int j = 1; j <= i; j++) {
                System.out.print("*");
            }
            System.out.println();
        }
    }

    // 沙漏图案
    public static void printHourglass(int n) {
        System.out.println("\nHourglass Pattern:");
        // 上半部分
        for (int i = n; i >= 1; i--) {
            for (int j = 1; j <= n - i; j++) {
                System.out.print(" ");
            }
            for (int j = 1; j <= 2 * i - 1; j++) {
                System.out.print("*");
            }
            System.out.println();
        }

        // 下半部分
        for (int i = 2; i <= n; i++) {
            for (int j = 1; j <= n - i; j++) {
                System.out.print(" ");
            }
            for (int j = 1; j <= 2 * i - 1; j++) {
                System.out.print("*");
            }
            System.out.println();
        }
    }

    // 锯齿图案
    public static void printZigzag(int rows, int cols) {
        System.out.println("\nZigzag Pattern:");
        for (int i = 1; i <= rows; i++) {
            for (int j = 1; j <= cols; j++) {
                if ((i + j) % 4 == 0 || (i == 2 && j % 4 == 0)) {
                    System.out.print("*");
                } else {
                    System.out.print(" ");
                }
            }
            System.out.println();
        }
    }

    // 螺旋图案
    public static void printSpiral(int n) {
        System.out.println("\nSpiral Pattern:");
        int[][] matrix = new int[n][n];
        int num = 1;
        int top = 0, bottom = n - 1, left = 0, right = n - 1;

        while (top <= bottom && left <= right) {
            // 填充顶行
            for (int i = left; i <= right; i++) {
                matrix[top][i] = num++;
            }
            top++;

            // 填充右列
            for (int i = top; i <= bottom; i++) {
                matrix[i][right] = num++;
            }
            right--;

            // 填充底行
            if (top <= bottom) {
                for (int i = right; i >= left; i--) {
                    matrix[bottom][i] = num++;
                }
                bottom--;
            }

            // 填充左列
            if (left <= right) {
                for (int i = bottom; i >= top; i--) {
                    matrix[i][left] = num++;
                }
                left++;
            }
        }

        // 输出螺旋矩阵
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                System.out.printf("%3d ", matrix[i][j]);
            }
            System.out.println();
        }
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        System.out.print("Enter the size for patterns: ");
        int size = scanner.nextInt();

        printButterfly(size);
        printHourglass(size);
        printZigzag(5, 15);
        printSpiral(size);

        scanner.close();
    }
}
```

## 图案逻辑拆解

### **核心概念：**

1. **外层循环**：控制行数
2. **内层循环**：控制每一行中的元素
   - 用于对齐的空格
   - 构成图案的字符/数字
3. **数学关系**：
   - 第 `i` 行输出 `i` 个元素（直角三角形）
   - 第 `i` 行输出 `2*i-1` 个元素（金字塔）
   - 居中需要 `n-i` 个空格

### **常用公式：**

| 图案类型   | 星号/元素数   | 空格数 |
| ---------- | ------------- | ------ |
| 直角三角形 | `i`           | `0`    |
| 左三角形   | `i`           | `n-i`  |
| 金字塔     | `2*i-1`       | `n-i`  |
| 倒置金字塔 | `2*(n-i+1)-1` | `i-1`  |

## 练习题

1. **修改现有图案**，改用不同的字符
2. **创建自定义图案**，组合多种形状
3. **添加彩色图案**，使用控制台颜色
4. **实现用户输入**，自定义图案
5. **创建动画图案**，使其随时间变化

## 时间复杂度与空间复杂度

- **时间复杂度**：大多数图案为 O(n²)
- **空间复杂度**：纯打印为 O(1)，基于矩阵的图案为 O(n²)

图案打印是掌握嵌套循环、逻辑思维，以及理解数学与编程之间关系的好方法！
